cmake_minimum_required(VERSION 3.24)
project(FSEProbe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FSEPROBE_PORTABLE_TESTS_ONLY "Build only portable core and Python tests, never the Windows app" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
find_package(Python3 3.10 COMPONENTS Interpreter REQUIRED)
add_executable(probe_core_tests tests/core_tests.cpp src/scene.cpp)
target_include_directories(probe_core_tests PRIVATE src)
if(MSVC)
  target_compile_options(probe_core_tests PRIVATE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(probe_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME core COMMAND probe_core_tests)
add_test(NAME python COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -v)
set_tests_properties(python PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 120)
set_tests_properties(core PROPERTIES TIMEOUT 30)
if(FSEPROBE_PORTABLE_TESTS_ONLY)
  message(STATUS "Portable tests only: NOT a Windows build or FSE hardware test")
  return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Application requires Windows x64 MSVC. On Linux use -DFSEPROBE_PORTABLE_TESTS_ONLY=ON")
endif()
# Locate the SDK shader compiler. Do not compile shaders at application runtime.
file(GLOB SDK_BIN_ROOTS "C:/Program Files (x86)/Windows Kits/10/bin/*")
list(SORT SDK_BIN_ROOTS COMPARE NATURAL ORDER DESCENDING)
set(FXC_HINTS)
foreach(root IN LISTS SDK_BIN_ROOTS)
  list(APPEND FXC_HINTS "${root}/x64")
endforeach()
find_program(FXC_EXECUTABLE NAMES fxc.exe HINTS ${FXC_HINTS})
if(NOT FXC_EXECUTABLE)
  message(FATAL_ERROR "Windows SDK fxc.exe missing. Install SDK or set -DFXC_EXECUTABLE=absolute-path")
endif()
message(STATUS "Using shader compiler: ${FXC_EXECUTABLE}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(SHADER_HEADERS)
foreach(entry BackgroundVS BackgroundPS HudVS HudPS)
  if(entry MATCHES "VS$")
    set(profile vs_5_0)
  else()
    set(profile ps_5_0)
  endif()
  set(out "${CMAKE_CURRENT_BINARY_DIR}/generated/${entry}.h")
  add_custom_command(OUTPUT "${out}"
    COMMAND "${FXC_EXECUTABLE}" /nologo /WX /Ges /O3 /T ${profile} /E ${entry}
      /Fh "${out}" /Vn "g_${entry}" "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders.hlsl"
    DEPENDS src/shaders.hlsl VERBATIM)
  list(APPEND SHADER_HEADERS "${out}")
endforeach()
add_executable(FSEProbe WIN32 src/main.cpp src/graphics.cpp src/diagnostic.cpp src/scene.cpp ${SHADER_HEADERS})
target_include_directories(FSEProbe PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(FSEProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(FSEProbe PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
target_link_libraries(FSEProbe PRIVATE d3d11 dxgi user32 shell32)
target_link_options(FSEProbe PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/application.manifest")
add_test(NAME app_self_test COMMAND FSEProbe --self-test)
set_tests_properties(app_self_test PROPERTIES TIMEOUT 20)
install(TARGETS FSEProbe RUNTIME DESTINATION bin)
